Part 1

🧩Declaring functions

A function is a named, reusable piece of code. It takes parameters, does some work and returns a value. A function without a return returns undefined.

Declaring and calling functions · ▶ Runs
function add(a, b) {
  return a + b;
}
console.log(add(2, 3));
console.log(add(2));          // missing arguments are undefined
console.log(add(2, 3, 99));   // extra arguments are ignored

function logIn(user, password = "Secret12", remember = false) {
  return `${user} / ${password} / remember=${remember}`;
}
console.log(logIn("asha"));
console.log(logIn("ravi", undefined, true));   // undefined uses the default
console.log(logIn("mei", null));               // null does not
Output
5
NaN
5
asha / Secret12 / remember=false
ravi / Secret12 / remember=true
mei / null / remember=false
  • JavaScript does not check how many arguments you pass. Missing ones are undefined and extra ones are ignored.
  • Default values replace undefined only. Passing null keeps null.

Rest, spread and options objects

Rest parameters, spread and options objects · ▶ Runs
function total(label, ...prices) {     // rest: gathers the remaining arguments into an array
  const sum = prices.reduce((s, p) => s + p, 0);
  return `${label}: ${sum}`;
}
console.log(total("cart", 100, 250, 75));

const prices = [10, 20, 30];
console.log(Math.max(...prices));       // spread: spreads an array into arguments
console.log(total("again", ...prices, 40));

function createUser({ name, role = "viewer", active = true } = {}) {   // destructured options
  return { name, role, active };
}
console.log(createUser({ name: "Asha", role: "admin" }));
console.log(createUser());
Output
cart: 425
30
again: 100
{ name: 'Asha', role: 'admin', active: true }
{ name: undefined, role: 'viewer', active: true }
When a function has more than two or three parameters, take one options object and destructure it. Calls read like labels (launch({ headless: true, slowMo: 50 })) and the order stops mattering. Playwright, Cypress and Jest all configure this way.
Part 2

🏹Function expressions and arrow functions

🧪 QA essentialPlaywrightJestCypress

A function can also be written as an expression and stored in a variable. The shorter arrow function syntax is the most common form in modern code and in test files.

Function expressions and arrow functions · ▶ Runs
const square = function (n) { return n * n; };   // function expression
const cube = n => n * n * n;                       // arrow, one parameter, implicit return
const greet = (first, last) => `Hi ${first} ${last}`;
const makeUser = name => ({ name, createdAt: "today" });   // wrap an object in ( )
const log = () => { console.log("no parameters, a block body"); };

console.log(square(4), cube(3));
console.log(greet("Asha", "Rao"));
console.log(makeUser("Ravi"));
log();
Output
16 27
Hi Asha Rao
{ name: 'Ravi', createdAt: 'today' }
no parameters, a block body
FormNotes
function name() {}Hoisted, so it can be called before its line. Has its own this.
const f = function () {}Not hoisted. Has its own this.
const f = () => {}Not hoisted. No own this or arguments: it uses the ones around it. Cannot be called with new.
Test files are mostly arrow functions: test("adds to cart", async ({ page }) => { ... }). The exceptions are Mocha and Cypress code that uses this, covered below.
Part 3

🔭Scope, hoisting and the temporal dead zone

Scope decides where a name can be seen. JavaScript looks for a name in the current block, then in each enclosing block and function, out to the global scope. This chain is fixed by where the code is written, which is called lexical scope.

Global, function and block scope · ▶ Runs
const appName = "Shop";                 // global scope

function checkout() {
  const total = 499;                    // function scope
  if (total > 100) {
    const message = "big order";        // block scope
    console.log(appName, total, message);
  }
  console.log(typeof message);          // not visible outside its block
}
checkout();
console.log(typeof total);              // not visible outside the function

let level = "global";
function shadow() {
  let level = "inside";                 // a new variable that hides the outer one
  console.log(level);
}
shadow();
console.log(level);
Output
Shop 499 big order
undefined
undefined
inside
global

Hoisting

Before running a scope, the engine registers its declarations. Function declarations are ready at once. var exists but is undefined. let and const exist but cannot be touched until their line runs. That gap is the temporal dead zone.

Hoisting and the temporal dead zone · ▶ Runs
console.log(sayHi());               // function declarations are hoisted: this works
function sayHi() { return "hi"; }

console.log(typeof oldStyle);       // var is hoisted but undefined
var oldStyle = 1;

try {
  console.log(modern);             // let and const are in the "temporal dead zone"
} catch (err) {
  console.log(err.name + ": " + err.message);
}
let modern = 2;
Output
hi
undefined
ReferenceError: Cannot access 'modern' before initialization
Declare variables at the top of the block where you need them, with const or let. Then hoisting never surprises you.
Part 4

🎒Closures

🧪 QA essentialJestMochaPlaywright

A closure is a function together with the variables it could see when it was created. Those variables stay alive as long as the function does, even after the outer function has returned. Closures give you private state without classes.

A closure keeps its variables alive · ▶ Runs
function makeCounter(start = 0) {
  let count = start;               // private: only the inner functions can see it
  return {
    next: () => ++count,
    reset: () => { count = start; },
    get value() { return count; },
  };
}

const visits = makeCounter();
const retries = makeCounter(10);
visits.next(); visits.next();
retries.next();
console.log(visits.value, retries.value);
visits.reset();
console.log(visits.value);
console.log(visits.count);         // there is no way to reach the variable directly
Output
2 11
0
undefined

Each call to makeCounter creates a fresh count, so visits and retries never interfere.

The classic loop closure bug · ▶ Runs
const viaVar = [];
for (var i = 0; i < 3; i++) viaVar.push(() => i);
console.log(viaVar.map(f => f()));    // one shared i, already 3

const viaLet = [];
for (let j = 0; j < 3; j++) viaLet.push(() => j);
console.log(viaLet.map(f => f()));    // let makes a new j for every turn
Output
[ 3, 3, 3 ]
[ 0, 1, 2 ]
With var, every function created in the loop shares one variable, which has its final value by the time they run. let creates a new binding for each turn. This was a famous interview question before let existed.
Closures are everywhere in test code: a let page declared in a describe block and set in beforeEach is captured by every test inside. Test data factories, custom wait helpers and Page Object methods all rely on them.
Part 5

👉this, call, apply and bind

🧪 QA essentialMochaCypressMocha this.timeout and Cypress aliases on this only work inside function, not arrow functions.

this is decided by how a function is called, not where it is written. Arrow functions are the exception: they never get their own this and use the one from the surrounding code.

Callthis is
obj.method()obj, the object before the dot
fn()undefined in strict code (the global object in old sloppy code)
fn.call(x, a, b), fn.apply(x, [a, b])x, chosen by you for this one call
fn.bind(x)a new function where this is always x
new Fn()the new object being created
arrow functionwhatever this was where the arrow was written
How this is decided · ▶ Runs
const cart = {
  owner: "Asha",
  items: ["pen", "book"],
  describe() {
    return `${this.owner} has ${this.items.length} items`;    // this is the object before the dot
  },
  describeLater() {
    return this.items.map(item => `${this.owner}: ${item}`);  // arrows use the outer this
  },
};
console.log(cart.describe());
console.log(cart.describeLater());

const loose = cart.describe;          // the method is taken off the object
try {
  console.log(loose());               // no object before the dot, so this is undefined
} catch (err) {
  console.log(err.message);
}

const bound = cart.describe.bind(cart);
console.log(bound());
console.log(cart.describe.call({ owner: "Ravi", items: [1, 2, 3] }));
Output
Asha has 2 items
[ 'Asha: pen', 'Asha: book' ]
Cannot read properties of undefined (reading 'owner')
Asha has 2 items
Ravi has 3 items

Why test runners care

Mocha puts helpers such as this.timeout() and this.retries() on the test's this. Cypress stores aliases on it too (this.user after cy.wrap(user).as("user")). Both only work when the test is a normal function. An arrow test borrows this from the code around it, and below, where describe is an arrow too, that is the module and not Mocha.

Mocha needs function, not arrows, to use this · 🧪 Tests run, one fails on purpose

}

Part 6

🧠Higher order functions

🧪 QA essentialPlaywrightCypress

A higher order function takes a function as an argument or returns one. You already used one: array.map(fn). They let you write the shape of a task once and plug in the details.

Higher order functions · ▶ Runs
// Functions are values: store them, pass them, return them.
const rules = {
  required: v => v !== "" || "is required",
  email: v => /^[^@\s]+@[^@\s]+\.\w+$/.test(v) || "must be an email",
  min: n => v => v.length >= n || `must be at least ${n} characters`,   // returns a function
};

function validate(value, ...checks) {
  const errors = [];
  for (const check of checks) {
    const result = check(value);
    if (result !== true) errors.push(result);
  }
  return errors;
}

console.log(validate("asha@example.com", rules.required, rules.email));
console.log(validate("ab", rules.required, rules.email, rules.min(3)));
console.log(validate("", rules.required));
Output
[]
[ 'must be an email', 'must be at least 3 characters' ]
[ 'is required' ]

rules.min(3) returns a new function that remembers 3. Returning configured functions like this is the basis of matchers, middleware and plugins.

Memoize and debounce, two closures you will meet · ▶ Runs
function memoize(fn) {
  const cache = new Map();
  return arg => {
    if (!cache.has(arg)) cache.set(arg, fn(arg));
    return cache.get(arg);
  };
}

let calls = 0;
const slowSquare = n => { calls++; return n * n; };
const fastSquare = memoize(slowSquare);
console.log(fastSquare(9), fastSquare(9), fastSquare(4), fastSquare(9));
console.log("real calls:", calls);

function debounce(fn, wait) {
  let timer;
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => fn(...args), wait);
  };
}
const search = debounce(text => console.log("searching for", text), 30);
search("p"); search("ph"); search("pho"); search("phone");   // typed quickly
Output
81 81 16 81
real calls: 2
searching for phone
  • memoize remembers results so a slow, pure function runs once per input.
  • debounce waits until calls stop for a while, then runs once. Search boxes use it so the server gets one request instead of one per key press.
Debounced inputs explain a classic test failure: the test types "phone" and checks the results at once, before the debounce timer fires. Assert on the results with an auto-waiting check (await expect(list).toHaveCount(3)) instead of reading them immediately.
Part 7

🪆Recursion and the call stack

A recursive function calls itself on a smaller piece of the problem. It needs a base case that stops the calls. Every call waits on the call stack until the one it made returns.

Recursion and the call stack · ▶ Runs
function factorial(n) {
  if (n <= 1) return 1;          // base case: stops the recursion
  return n * factorial(n - 1);   // recursive case: a smaller problem
}
console.log(factorial(5));

// Recursion fits nested data, such as a site menu
const menu = { label: "Home", children: [
  { label: "Shop", children: [{ label: "Phones" }, { label: "Laptops", children: [{ label: "Gaming" }] }] },
  { label: "Help" },
] };

function allLabels(node, depth = 0) {
  const lines = ["  ".repeat(depth) + node.label];
  for (const child of node.children ?? []) lines.push(...allLabels(child, depth + 1));
  return lines;
}
console.log(allLabels(menu).join("\n"));

function forever(n) { return forever(n + 1); }   // no base case
try { forever(0); } catch (err) { console.log(err.name + ": " + err.message); }
Output
120
Home
  Shop
    Phones
    Laptops
      Gaming
  Help
RangeError: Maximum call stack size exceeded

Recursion is natural for nested data: menus, folder trees, JSON with nested objects, the DOM itself. Without a base case, the stack fills up and you get RangeError: Maximum call stack size exceeded.

Part 8

🎬Real world scenario: a test data factory

🧪 QA essential
The situationYour sign up tests keep failing with "email already registered" because every test uses test@example.com. You need valid, unique users by default, and an easy way for one test to ask for an admin or an unverified user.
A test data factory built on a closure · ▶ Runs
// A test data builder: unique, valid data by default, overridable when a test needs something special.
function userFactory(prefix = "qa") {
  let seq = 0;                                  // private counter, kept by the closure
  return function build(overrides = {}) {
    seq++;
    return {
      id: seq,
      name: `Test User ${seq}`,
      email: `${prefix}.user${seq}@example.com`,
      role: "customer",
      verified: true,
      ...overrides,                             // the test's own values win
    };
  };
}

const buildUser = userFactory("checkout");
const normal = buildUser();
const admin = buildUser({ role: "admin" });
const unverified = buildUser({ verified: false, email: "not-an-email" });
console.table([normal, admin, unverified]);

const other = userFactory("search")();          // a separate counter for another suite
console.log(other.email);
Output
┌─────────┬────┬───────────────┬──────────────────────────────┬────────────┬──────────┐
│ (index) │ id │ name          │ email                        │ role       │ verified │
├─────────┼────┼───────────────┼──────────────────────────────┼────────────┼──────────┤
│ 0       │ 1  │ 'Test User 1' │ 'checkout.user1@example.com' │ 'customer' │ true     │
│ 1       │ 2  │ 'Test User 2' │ 'checkout.user2@example.com' │ 'admin'    │ true     │
│ 2       │ 3  │ 'Test User 3' │ 'not-an-email'               │ 'customer' │ false    │
└─────────┴────┴───────────────┴──────────────────────────────┴────────────┴──────────┘
search.user1@example.com
  • The closure keeps a private seq counter per factory, so emails never repeat within a run.
  • The object spread ...overrides lets each test change only what it cares about. The test reads as "a user, but unverified".
  • Separate factories have separate counters, so suites do not depend on each other.
The same pattern builds fixtures for unit tests of the app code itself. Libraries such as Fishery and @faker-js/faker grow it further with realistic random values.
