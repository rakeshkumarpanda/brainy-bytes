Part 1

🧬Objects and prototypes

Every object in JavaScript has a hidden link to another object called its prototype. When you read a property the object does not have, JavaScript follows that link and looks there, then on that object's prototype, and so on. This list of links is the prototype chain. It ends at Object.prototype, whose prototype is null.

Looking up the prototype chain · ▶ Runs
const animal = {
  breathe() { return `${this.name} breathes`; },
};
const dog = Object.create(animal);        // dog's prototype is animal
dog.name = "Rex";
dog.bark = function () { return "Woof"; };

console.log(dog.bark());                   // found on dog itself
console.log(dog.breathe());                // not on dog, found one step up
console.log(Object.getPrototypeOf(dog) === animal);
console.log(Object.hasOwn(dog, "breathe"), "breathe" in dog);

const arr = [1, 2];
console.log(Object.getPrototypeOf(arr) === Array.prototype);
console.log(Object.getPrototypeOf(Array.prototype) === Object.prototype);
console.log(Object.getPrototypeOf(Object.prototype));   // the end of every chain

Before the class keyword existed, shared methods were put on a constructor function's prototype object and objects were made with new. Classes still work exactly this way underneath.

Constructor functions and new · ▶ Runs
// Before 2015 this is how "classes" were written. You still see it in older libraries.
function User(name) {
  this.name = name;
}
User.prototype.greet = function () {
  return `Hi, ${this.name}`;
};

const a = new User("Asha");
const b = new User("Ravi");
console.log(a.greet(), b.greet());
console.log(a.greet === b.greet);          // one shared copy on the prototype
console.log(a instanceof User, Object.getPrototypeOf(a) === User.prototype);
Output
Hi, Asha Hi, Ravi
true
true true
new User("Asha") does four things: it creates an empty object, links it to User.prototype, runs User with this set to the new object, and returns it.
Part 2

🏛️Classes

🧪 QA essentialPage objects, API clients, fixtures and custom reporters in every automation framework are classes.

A class is a template for objects that share the same shape and methods. It has a constructor that sets up each new object, methods shared through the prototype, fields that every object gets, getters and setters that look like properties, and static members that belong to the class itself.

A class with fields, a getter and a static method · ▶ Runs
class Cart {
  static currency = "INR";                 // belongs to the class, not each cart
  items = [];                              // a public field, set up for every new cart

  constructor(owner) {
    this.owner = owner;
  }

  add(name, price, qty = 1) {
    this.items.push({ name, price, qty });
    return this;                           // returning this allows chaining
  }

  get total() {                            // read like a property, computed each time
    return this.items.reduce((sum, i) => sum + i.price * i.qty, 0);
  }

  static fromJSON(json) {
    const data = JSON.parse(json);
    const cart = new Cart(data.owner);
    for (const i of data.items) cart.add(i.name, i.price, i.qty);
    return cart;
  }
}

const cart = new Cart("Asha").add("Pen", 20, 3).add("Book", 250);
console.log(cart.total, Cart.currency);
console.log(typeof Cart, Object.getOwnPropertyNames(Cart.prototype));

const copy = Cart.fromJSON('{"owner":"Ravi","items":[{"name":"Mug","price":199,"qty":2}]}');
console.log(copy.owner, copy.total);

Two traps

Class bodies are always in strict mode. A method pulled off its object and called on its own gets this as undefined. An arrow function stored in a field keeps its object. Classes are also not usable before the line that declares them.

Two class traps: lost this and no hoisting · ▶ Runs
class Counter {
  count = 0;
  increment() { this.count++; }
  incrementArrow = () => { this.count++; };   // a field holding an arrow, bound per object
}

const c = new Counter();
const { increment, incrementArrow } = c;
incrementArrow();
console.log(c.count);
try {
  increment();                             // class bodies are strict, so this is undefined
} catch (e) {
  console.log(e.constructor.name + ":", e.message);
}

try {
  new Shape();                             // classes are not hoisted like functions
} catch (e) {
  console.log(e.constructor.name + ":", e.message);
}
class Shape {}
Output
1
TypeError: Cannot read properties of undefined (reading 'count')
ReferenceError: Cannot access 'Shape' before initialization
Part 3

🔒Private fields and encapsulation

A field or method whose name starts with # is private. Only code inside the class body can read or change it. It does not show up in Object.keys, in JSON or in the console of other code, and any attempt to read it from outside is a syntax error. This is real privacy, not a naming habit like the old _balance.

Private fields and methods · ▶ Runs
class BankAccount {
  #balance = 0;                            // private: only code inside the class can see it
  static #count = 0;

  constructor(owner) {
    this.owner = owner;
    BankAccount.#count++;
  }

  deposit(amount) {
    this.#check(amount);
    this.#balance += amount;
    return this.#balance;
  }

  #check(amount) {                         // a private method
    if (!(amount > 0)) throw new RangeError(`Invalid amount: ${amount}`);
  }

  get balance() { return this.#balance; }
  static get count() { return BankAccount.#count; }
  static isAccount(obj) { return #balance in obj; }   // a safe brand check
}

const acc = new BankAccount("Mei");
acc.deposit(500);
console.log(acc.balance, BankAccount.count);
console.log(Object.keys(acc), JSON.stringify(acc));
console.log(BankAccount.isAccount(acc), BankAccount.isAccount({ balance: 1 }));
for (const attempt of [() => { acc.balance = 1_000_000; }, () => acc.deposit(-5)]) {
  try {
    attempt();
  } catch (e) {
    console.log(e.name + ":", e.message);
  }
}
console.log(acc.balance);
Output

A static block runs once when the class is defined. It is useful for set up that needs a few statements, such as picking a base URL from an environment variable.

Static initialisation blocks · ▶ Runs
class Config {
  static env;
  static baseURL;
  static {                                 // runs once when the class is defined
    Config.env = process.env.TEST_ENV ?? "staging";
    Config.baseURL = { staging: "https://staging.shop.example", prod: "https://shop.example" }[Config.env];
  }
}
console.log(Config.env, Config.baseURL);
Output
staging https://staging.shop.example
Public and private fields, private methods, #x in obj and static blocks arrived in ES2022. Every current browser and Node.js support them.
Part 4

🌳Inheritance with extends and super

🧪 QA essentialA BasePage that every page object extends, and custom errors with a status and a cause, are standard in test frameworks.

class Dog extends Animal makes Animal.prototype the prototype of Dog.prototype. A Dog gets every Animal method and can override any of them. Inside the subclass, super(...) calls the parent constructor, and super.method() calls the parent version of a method.

extends and super · ▶ Runs
class Animal {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} makes a sound`; }
  toString() { return `[${this.constructor.name} ${this.name}]`; }
}

class Dog extends Animal {
  constructor(name, breed) {
    super(name);                           // must run before this is used
    this.breed = breed;
  }
  speak() {                                // overrides the parent method
    return `${super.speak()}, then barks`;  // and can still call it
  }
}

const d = new Dog("Rex", "beagle");
console.log(d.speak());
console.log(`${d}`);
console.log(d instanceof Dog, d instanceof Animal, d instanceof Object);
console.log(Object.getPrototypeOf(Dog.prototype) === Animal.prototype);
Output
Rex makes a sound, then barks
[Dog Rex]
true true true
true
In a subclass constructor, super() must run before the first use of this. Forgetting it throws a ReferenceError as soon as the class is used.

Custom errors

Extending Error gives errors a clear type you can check with instanceof, extra data such as a status code, and a cause that keeps the original error. Test frameworks print the cause too, which makes failures much easier to read.

Custom error classes · ▶ Runs
class HttpError extends Error {
  constructor(status, url, options) {
    super(`Request to ${url} failed with ${status}`, options);
    this.name = "HttpError";
    this.status = status;
  }
  get retryable() { return this.status >= 500 || this.status === 429; }
}

class NotFoundError extends HttpError {
  constructor(url) {
    super(404, url);
    this.name = "NotFoundError";
  }
}

function load(status) {
  try {
    throw new TypeError("network down");
  } catch (low) {
    if (status === 404) throw new NotFoundError("/api/users/9");
    throw new HttpError(status, "/api/users", { cause: low });   // keep the original error
  }
}

for (const status of [404, 503]) {
  try {
    load(status);
  } catch (e) {
    console.log(e.name, "|", e.message, "| retry:", e.retryable);
    console.log("  instanceof HttpError:", e instanceof HttpError, "| cause:", e.cause?.message);
  }
}
Output
NotFoundError | Request to /api/users/9 failed with 404 | retry: false
  instanceof HttpError: true | cause: undefined
HttpError | Request to /api/users failed with 503 | retry: true
  instanceof HttpError: true | cause: network down
Part 5

🧩Composition over inheritance

Deep class trees get hard to change: a fix in a parent class affects every child. Many teams prefer composition, where an object is built from small parts it holds or mixes in. A mixin is a function that takes a class and returns a new class with extra abilities. Passing a helper into the constructor, called dependency injection, lets tests replace it with a fake.

Mixins and composition · ▶ Runs
// Instead of a deep tree of subclasses, build objects from small abilities.
const canLog = base => class extends base {
  log(msg) { (this.logs ??= []).push(msg); return this; }
};
const canRetry = base => class extends base {
  async retry(fn, times = 3) {
    for (let i = 1; ; i++) {
      try { return await fn(i); } catch (e) {
        this.log?.(`attempt ${i} failed: ${e.message}`);
        if (i === times) throw e;
      }
    }
  }
};

class Client {}
class ApiClient extends canRetry(canLog(Client)) {}

const api = new ApiClient();
const result = await api.retry(n => {
  if (n < 3) throw new Error("timeout");
  return "200 OK";
});
console.log(result);
console.log(api.logs);

// Composition with plain objects: the client HAS a logger instead of BEING one
class Uploader {
  constructor(logger = console) { this.logger = logger; }
  upload(file) { this.logger.log(`uploading ${file}`); return true; }
}
const messages = [];
new Uploader({ log: m => messages.push(m) }).upload("report.pdf");
console.log(messages);
Output
200 OK
[ 'attempt 1 failed: timeout', 'attempt 2 failed: timeout' ]
[ 'uploading report.pdf' ]
Inheritance: "is a"A CheckoutPage is a BasePage. Good for one shallow level of shared behaviour.
Composition: "has a"A CheckoutPage has a Header and a CartSummary component. Good for parts that appear on many pages.
Part 6

📄The Page Object Model

🧪 QA essentialPlaywrightSeleniumCypressWebdriverIOAlmost every UI automation framework you join will be organised as page objects.

The Page Object Model (POM) is the most common design pattern in UI automation. Each page or component of the app gets a class. The class holds the locators and offers methods with business names such as loginAs or addToCart. Tests then call those methods and never touch selectors directly.

A page object for a login form · ▶ Runs in the browser
📄 page.html
<!doctype html>
<html>
<body>
  <form id="login">
    <label>Email <input id="email" type="email"></label>
    <label>Password <input id="password" type="password"></label>
    <button type="submit" data-test="sign-in">Sign in</button>
    <p class="error" hidden></p>
  </form>

document.querySelector("#login").addEventListener("submit", e => {
  e.preventDefault();
  const email = document.querySelector("#email").value;
  const pass = document.querySelector("#password").value;
  const error = document.querySelector(".error");
  if (pass !== "s3cret") {
    error.textContent = "Wrong email or password";
    error.hidden = false;
    return;
  }
  error.hidden = true;
  const welcome = document.querySelector("#welcome");
  welcome.textContent = `Hello, ${email.split("@")[0]}`;
  welcome.hidden = false;
});
</script>
</body>
</html>
📄 script.js
// A page object hides the selectors behind methods with business names.
class LoginPage {
  constructor(doc) { this.doc = doc; }

  get email() { return this.doc.querySelector("#email"); }
  get password() { return this.doc.querySelector("#password"); }
  get submit() { return this.doc.querySelector('[data-test="sign-in"]'); }
  get error() { return this.doc.querySelector(".error"); }
  get welcome() { return this.doc.querySelector("#welcome"); }

  loginAs(email, password) {
    this.email.value = email;
    this.password.value = password;
    this.submit.click();
    return this;
  }
}

const page = new LoginPage(document);
page.loginAs("asha@shop.example", "wrong");
console.log("error shown:", !page.error.hidden, "|", page.error.textContent);

page.loginAs("asha@shop.example", "s3cret");
console.log("error shown:", !page.error.hidden, "| welcome:", page.welcome.textContent);
Output
error shown: true | Wrong email or password
error shown: false | welcome: Hello, asha

The idea is identical in every tool. Only the way you find and click elements changes.

The same page object in Playwright · Runs for real in chapter 15
// pages/login.page.js (Playwright)
export class LoginPage {
  constructor(page) {
    this.page = page;
    this.email = page.getByLabel("Email");
    this.password = page.getByLabel("Password");
    this.submit = page.getByRole("button", { name: "Sign in" });
    this.error = page.locator(".error");
  }
  async goto() {
    await this.page.goto("/login");
  }
  async loginAs(email, password) {
    await this.email.fill(email);
    await this.password.fill(password);
    await this.submit.click();
  }
}

// tests/login.spec.js
import { test, expect } from "@playwright/test";
import { LoginPage } from "../pages/login.page.js";

test("wrong password shows an error", async ({ page }) => {
  const login = new LoginPage(page);
  await login.goto();
  await login.loginAs("asha@shop.example", "wrong");
  await expect(login.error).toHaveText("Wrong email or password");
});
What happens

The test reads like the steps a manual tester writes. When the button text changes, only the page object changes.

The same page object in Selenium · Runs for real in chapter 15
// pages/LoginPage.js (Selenium WebDriver)
const { By, until } = require("selenium-webdriver");

class LoginPage {
  static EMAIL = By.id("email");
  static PASSWORD = By.id("password");
  static SUBMIT = By.css('[data-test="sign-in"]');
  static ERROR = By.css(".error");

  constructor(driver) { this.driver = driver; }

  async open(baseUrl) {
    await this.driver.get(`${baseUrl}/login`);
  }
  async loginAs(email, password) {
    await this.driver.findElement(LoginPage.EMAIL).sendKeys(email);
    await this.driver.findElement(LoginPage.PASSWORD).sendKeys(password);
    await this.driver.findElement(LoginPage.SUBMIT).click();
  }
  async errorText() {
    const el = await this.driver.wait(until.elementLocated(LoginPage.ERROR), 5000);
    return el.getText();
  }
}
module.exports = { LoginPage };
What happens

Locators live in static fields, and every method is async because each WebDriver call talks to the browser.

Rules that keep page objects healthy: one class per page or component, no assertions inside page objects (they belong in tests), methods return useful values or other page objects, and locators prefer roles, labels and data-test attributes over long CSS paths.

🎬Real world scenario: test data and an API client

🧪 QA essential
The situationYour API tests create many users with small differences, and every test repeats the same HTTP code. You want readable test data and one place that knows how to call the API.

A builder class starts from sensible defaults, and each test changes only what matters to it. Every built user gets a fresh email, so tests never clash.

A test data builder · ▶ Runs
class UserBuilder {
  #data = { name: "Test User", email: "user@test.example", role: "customer", verified: true, address: { city: "Pune" } };
  static #seq = 0;

  withName(name) { this.#data.name = name; return this; }
  withRole(role) { this.#data.role = role; return this; }
  unverified() { this.#data.verified = false; return this; }
  inCity(city) { this.#data.address = { ...this.#data.address, city }; return this; }

  build() {
    const id = ++UserBuilder.#seq;
    return structuredClone({ id, ...this.#data, email: `user${id}@test.example` });
  }
}

const admin = new UserBuilder().withName("Asha").withRole("admin").build();
const newbie = new UserBuilder().unverified().inCity("Bhubaneswar").build();
console.log(admin);
console.log(newbie);
console.log(admin.email !== newbie.email);

An API client base class owns the HTTP details and logging. Small subclasses add one method per endpoint. Because the transport is passed in, the same client works against a fake in unit tests and the real server in integration tests.

An API client with a base class and an injected transport · ▶ Runs
// One base class knows HTTP. Each area of the API gets a small subclass.
class ApiClient {
  #log = [];
  constructor(baseURL, transport) {
    this.baseURL = baseURL;
    this.transport = transport;            // injected, so tests can pass a fake
  }
  async request(method, path, body) {
    const res = await this.transport({ method, url: this.baseURL + path, body });
    this.#log.push(`${method} ${path} -> ${res.status}`);
    if (res.status >= 400) throw new Error(`${method} ${path} failed with ${res.status}`);
    return res.body;
  }
  get history() { return [...this.#log]; }
}

class UsersApi extends ApiClient {
  get(id) { return this.request("GET", `/users/${id}`); }
  create(user) { return this.request("POST", "/users", user); }
}

// A fake transport stands in for the network
const db = new Map([[1, { id: 1, name: "Asha" }]]);
async function fakeTransport({ method, url, body }) {
  const id = Number(url.split("/").pop());
  if (method === "GET") return db.has(id) ? { status: 200, body: db.get(id) } : { status: 404 };
  const user = { id: db.size + 1, ...body };
  db.set(user.id, user);
  return { status: 201, body: user };
}

const users = new UsersApi("https://api.shop.example", fakeTransport);
console.log(await users.get(1));
console.log(await users.create({ name: "Ravi" }));
await users.get(42).catch(e => console.log("Error:", e.message));
console.log(users.history);
Output

{ id: 2, name: 'Ravi' }
